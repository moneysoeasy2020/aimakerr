### 2.2 Component Deep Dive

#### A. Local Inference Server (vLLM / Ollama)

- **vLLM** for high-throughput batch workloads (PagedAttention, continuous batching). Ideal for A100/4090 rigs serving 20+ concurrent agent requests.
- **Ollama** for developer ergonomics and single-model agents. Exposes an OpenAI-compatible `/v1/chat/completions` endpoint — critical for drop-in router compatibility.
- **llama.cpp** for CPU/edge deployments and quantized (GGUF Q4_K_M) models.

**Key tuning parameters:**
| Parameter | vLLM | Ollama | Effect |
|---|---|---|---|
| `--max-model-len` | 8192 | `num_ctx` | Context window |
| `--gpu-memory-utilization` | 0.90 | — | VRAM headroom |
| `--max-num-seqs` | 32 | `OLLAMA_NUM_PARALLEL` | Concurrency |
| `--tensor-parallel-size` | N GPUs | — | Multi-GPU |

#### B. Async Agent Orchestration

The core pattern is **fan-out/fan-in** with bounded concurrency:

- `asyncio.Semaphore` caps in-flight requests to match GPU capacity.
- `httpx.AsyncClient` with connection pooling reuses TCP sockets to the local server.
- `tenacity` provides exponential backoff with jitter for transient GPU OOM / 503 responses.
- A **circuit breaker** trips to cloud fallback when local latency exceeds SLA.

#### C. Hybrid Router Logic

```python
# Pseudocode decision tree
if task.sensitivity == "high" or task.volume > threshold:
route = LOCAL
elif local_queue_depth > MAX_DEPTH or local_p95_latency > SLA:
route = CLOUD_FALLBACK
else:
route = LOCAL
```

This preserves data sovereignty for sensitive workloads while guaranteeing throughput under load.

#### D. AEO/GEO Traffic Interception Layer

AI answer engines (Perplexity, ChatGPT Search, Google AI Overviews, Claude) crawl structured surfaces. The pipeline must emit:

1. **`llms.txt`** — a curated markdown index at site root guiding LLM crawlers to canonical content.
2. **JSON-LD `FAQPage` / `HowTo` / `TechArticle` schema** — machine-parseable Q&A pairs the answer engine can quote verbatim.
3. **Semantic HTML** — `
`, `

` question-form headings, `` comparison data.
4. **Canonical answer blocks** — 40–60 word "answer-first" paragraphs immediately after each H2 (the AEO "snippet zone").

The agent pipeline generates these artifacts automatically from local LLM outputs, then publishes via a static site generator or headless CMS webhook.

### 2.3 Data Flow

1. **Ingest** → task queue (Redis / `asyncio.Queue`).
2. **Route** → hybrid router selects local or cloud endpoint.
3. **Infer** → async call with retry + semaphore.
4. **Validate** → Pydantic schema check on structured output.
5. **Publish** → AEO formatter emits JSON-LD + markdown + `llms.txt` fragment.

### 3.2 Async Agent Pipeline (`agent_pipeline.py`)

```python
"""
Production async agent pipeline for local LLM workflows.
- Bounded concurrency via Semaphore
- Retry with exponential backoff + jitter
- Hybrid local/cloud routing with circuit breaker
- Pydantic-validated structured output
"""
import asyncio
import json
import logging
import os
import time
from dataclasses import dataclass, field
from typing import Literal

import httpx
from pydantic import BaseModel, Field
from tenacity import (
retry, stop_after_attempt, wait_exponential_jitter,
retry_if_exception_type,
)

logging.basicConfig(
level=logging.INFO,
format='{"ts":"%(asctime)s","lvl":"%(levelname)s","msg":"%(message)s"}',
)
log = logging.getLogger("agent")

LOCAL_BASE = os.getenv("LOCAL_LLM_BASE", "http://localhost:8000/v1")
CLOUD_BASE = os.getenv("CLOUD_LLM_BASE", "https://api.openai.com/v1")
CLOUD_KEY = os.getenv("OPENAI_API_KEY", "")
LOCAL_MODEL = os.getenv("LOCAL_MODEL", "Qwen/Qwen2.5-7B-Instruct")
CLOUD_MODEL = os.getenv("CLOUD_MODEL", "gpt-4o-mini")

MAX_CONCURRENCY = int(os.getenv("MAX_CONCURRENCY", "16"))
LOCAL_P95_SLA_MS = int(os.getenv("LOCAL_P95_SLA_MS", "8000"))


# ---------- Structured output contract ----------
class AgentOutput(BaseModel):
answer: str = Field(..., min_length=20, max_length=1200)
confidence: float = Field(..., ge=0.0, le=1.0)
citations: list[str] = Field(default_factory=list)
faq: list[dict[str, str]] = Field(default_factory=list) # AEO payload


# ---------- Circuit breaker ----------
@dataclass
class CircuitBreaker:
failures: int = 0
threshold: int = 5
cooldown_s: float = 30.0
opened_at: float = 0.0
_latencies: list[float] = field(default_factory=list)

def record(self, latency_ms: float, ok: bool) -> None:
self._latencies.append(latency_ms)
self._latencies = self._latencies[-100:]
if ok:
self.failures = max(0, self.failures - 1)
else:
self.failures += 1
if self.failures >= self.threshold:
self.opened_at = time.time()
log.warning("circuit_open")

def is_open(self) -> bool:
if self.failures < self.threshold:
return False
if time.time() - self.opened_at > self.cooldown_s:
self.failures = 0
log.info("circuit_half_open")
return False
return True

def p95(self) -> float:
if not self._latencies:
return 0.0
s = sorted(self._latencies)
return s[int(len(s) * 0.95) - 1]


breaker = CircuitBreaker()
sem = asyncio.Semaphore(MAX_CONCURRENCY)


# ---------- Inference call ----------
class TransientError(Exception):
pass


@retry(
retry=retry_if_exception_type((TransientError, httpx.TransportError)),
wait=wait_exponential_jitter(initial=0.5, max=8.0, jitter=1.0),
stop=stop_after_attempt(4),
reraise=True,
)
async def _call_llm(
client: httpx.AsyncClient,
base: str,
model: str,
prompt: str,
api_key: str | None = None,
) -> dict:
headers = {"Content-Type": "application/json"}
if api_key:
headers["Authorization"] = f"Bearer {api_key}"

payload = {
"model": model,
"messages": [
{"role": "system", "content":
"You are a technical writer. Return STRICT JSON matching: "
'{"answer": str, "confidence": float, "citations": [str], '
'"faq": [{"q": str, "a": str}]}'},
{"role": "user", "content": prompt},
],
"temperature": 0.2,
"response_format": {"type": "json_object"},
}

t0 = time.perf_counter()
try:
r = await client.post(f"{base}/chat/completions",
json=payload, headers=headers, timeout=60.0)
if r.status_code in (429, 503):
raise TransientError(f"status {r.status_code}")
r.raise_for_status()
latency_ms = (time.perf_counter() - t0) * 1000
breaker.record(latency_ms, ok=True)
return r.json()
except (httpx.TransportError, TransientError):
breaker.record((time.perf_counter() - t0) * 1000, ok=False)
raise


async def run_agent(
client: httpx.AsyncClient,
prompt: str,
sensitivity: Literal["low", "high"] = "low",
) -> AgentOutput:
"""Route → infer → validate. Falls back to cloud on circuit-open."""
async with sem:
use_local = (sensitivity == "low"
and not breaker.is_open()
and breaker.p95() < LOCAL_P95_SLA_MS)

if use_local:
try:
raw = await _call_llm(client, LOCAL_BASE, LOCAL_MODEL, prompt)
log.info("route=local")
except Exception as e:
log.warning(f"local_failed:{e} route=cloud")
raw = await _call_llm(client, CLOUD_BASE, CLOUD_MODEL,
prompt, api_key=CLOUD_KEY)
else:
raw = await _call_llm(client, CLOUD_BASE, CLOUD_MODEL,
prompt, api_key=CLOUD_KEY)
log.info("route=cloud")

content = raw["choices"][0]["message"]["content"]
return AgentOutput.model_validate_json(content)


# ---------- Batch orchestrator ----------
async def run_batch(prompts: list[str], sensitivity="low") -> list[AgentOutput]:
limits = httpx.Limits(max_connections=MAX_CONCURRENCY * 2,
max_keepalive_connections=MAX_CONCURRENCY)
async with httpx.AsyncClient(limits=limits) as client:
tasks = [run_agent(client, p, sensitivity) for p in prompts]
results = await asyncio.gather(*tasks, return_exceptions=True)

ok, failed = [], []
for p, r in zip(prompts, results):
if isinstance(r, Exception):
failed.append((p, str(r)))
else:
ok.append(r)
log.info(f"batch_done ok={len(ok)} failed={len(failed)}")
return ok


if __name__ == "__main__":
prompts = [f"Explain async pipeline pattern #{i} for AEO content." for i in range(20)]
out = asyncio.run(run_batch(prompts))
print(json.dumps([o.model_dump() for o in out[:2]], indent=2))
```

### 3.3 AEO Publisher (`aeo_publisher.py`)

```python
"""
Emit AEO/GEO artifacts from agent outputs:
- JSON-LD FAQPage schema
- answer-first markdown blocks
- llms.txt fragment
"""
import json
from pathlib import Path
from agent_pipeline import AgentOutput

def to_jsonld(topic: str, out: AgentOutput, url: str) -> dict:
return {
"@context": "https://schema.org",
"@type": "FAQPage",
"mainEntity": [
{
"@type": "Question",
"name": item["q"],
"acceptedAnswer": {"@type": "Answer", "text": item["a"]},
}
for item in out.faq
],
"about": {"@type": "TechArticle", "headline": topic, "url": url},
}

def to_markdown(topic: str, out: AgentOutput) -> str:
lines = [f"## {topic}", "", out.answer, ""]
for item in out.faq:
lines += [f"### {item['q']}", item["a"], ""]
return "\n".join(lines)

def publish(topic: str, out: AgentOutput, out_dir: Path, url: str) -> None:
out_dir.mkdir(parents=True, exist_ok=True)
slug = topic.lower().replace(" ", "-")[:60]
(out_dir / f"{slug}.md").write_text(to_markdown(topic, out))
(out_dir / f"{slug}.jsonld").write_text(
json.dumps(to_jsonld(topic, out, url), indent=2))
# Append to llms.txt index
idx = out_dir / "llms.txt"
with idx.open("a") as f:
f.write(f"- [{topic}]({url}/{slug})\n")
```
